{{- /* Static tile map. OSM raster tiles are composed into one SVG, cropped to a window centred on the coordinates, so the card gets a real map with no JavaScript, no iframe and no key. The tiles are filtered to the site's palette in CSS; the marker and the attribution stay unfiltered. */ -}} {{- $lat := float .lat -}} {{- $lon := float .lon -}} {{- $z := 13 -}} {{- $size := 256.0 -}} {{- $n := math.Pow 2 $z -}} {{- $world := mul $size $n -}} {{- $x := mul (div (add $lon 180.0) 360.0) $world -}} {{- $phi := div (mul $lat math.Pi) 180.0 -}} {{- $merc := math.Log (add (math.Tan $phi) (div 1.0 (math.Cos $phi))) -}} {{- $y := mul (div (sub 1.0 (div $merc math.Pi)) 2.0) $world -}} {{- $w := 512.0 -}} {{- $h := 204.0 -}} {{- $x0 := sub $x (div $w 2.0) -}} {{- $y0 := sub $y (div $h 2.0) -}} {{- $txFrom := int (math.Floor (div $x0 $size)) -}} {{- $txTo := int (math.Floor (div (sub (add $x0 $w) 0.01) $size)) -}} {{- $tyFrom := int (math.Floor (div $y0 $size)) -}} {{- $tyTo := int (math.Floor (div (sub (add $y0 $h) 0.01) $size)) -}} {{- $tiles := int $n -}} {{- $cx := sub $x $x0 -}} {{- $cy := sub $y $y0 -}}
